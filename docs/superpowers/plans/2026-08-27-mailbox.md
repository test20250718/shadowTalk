```python
class MailCacheRepository:
    """邮箱客户端邮件缓存（独立于 processed_emails 分身同步去重表）。"""

    @staticmethod
    def upsert_batch(folder: str, emails: list[dict]) -> None:
        """批量 upsert 邮件到缓存（按 folder+message_id 去重）。"""
        if not emails:
            return
        with Database.transaction() as conn:
            for e in emails:
                conn.execute(
                    """INSERT INTO mail_cache
                       (folder, message_id, sender, sender_addr, subject,
                        body_text, received_at)
                       VALUES (?, ?, ?, ?, ?, ?, ?)
                       ON CONFLICT(folder, message_id) DO UPDATE SET
                           sender=excluded.sender,
                           sender_addr=excluded.sender_addr,
                           subject=excluded.subject,
                           body_text=excluded.body_text,
                           received_at=excluded.received_at,
                           fetched_at=CURRENT_TIMESTAMP""",
                    (folder,
                     e.get("message_id", ""),
                     e.get("sender", ""),
                     e.get("sender_addr", ""),
                     e.get("subject", ""),
                     e.get("body_text", ""),
                     e.get("received_at", ""))
                )

    @staticmethod
    def get_by_folder(folder: str, limit: int | None = None):
        """按文件夹获取邮件（received_at 降序，最新在前）。"""
        conn = Database.get_connection()
        sql = ("SELECT * FROM mail_cache WHERE folder=? "
               "ORDER BY received_at DESC, id DESC")
        if limit is not None:
            sql += f" LIMIT {int(limit)}"
        return conn.execute(sql, (folder,)).fetchall()

    @staticmethod
    def get_by_id(email_id: int):
        conn = Database.get_connection()
        return conn.execute(
            "SELECT * FROM mail_cache WHERE id=?", (email_id,)
        ).fetchone()

    @staticmethod
    def mark_read(email_id: int) -> None:
        with Database.transaction() as conn:
            conn.execute(
                "UPDATE mail_cache SET is_read=1 WHERE id=?", (email_id,)
            )

    @staticmethod
    def get_folders() -> list[str]:
        """返回有缓存数据的所有文件夹名。"""
        conn = Database.get_connection()
        rows = conn.execute(
            "SELECT DISTINCT folder FROM mail_cache ORDER BY folder"
        ).fetchall()
        return [r[0] for r in rows]
```

```python
# shadowtalk/ui/threads/mail_fetch_worker.py
"""
邮箱客户端 IMAP 拉取线程（QThread）。

与分身同步的 MailReceiveWorker 不同：本 worker 拉取指定文件夹的全部邮件，
转换为 dict 列表由 emails_fetched 信号发出，由调用方写入 mail_cache 表。
不匹配 session、不插回聊天。
"""
import logging
import email.utils
from datetime import datetime

from PySide6.QtCore import QThread, Signal

from shadowtalk.core.mail_receiver import MailReceiver

logger = logging.getLogger(__name__)


class MailFetchWorker(QThread):
    """邮箱客户端 IMAP 拉取线程。"""

    emails_fetched = Signal(list)   # list[dict]
    failed = Signal(str)            # reason

    def __init__(self, imap_config: dict, folder: str = "INBOX", parent=None):
        super().__init__(parent)
        self.imap_config = imap_config
        self.folder = folder
        self._cancelled = False

    def cancel(self):
        self._cancelled = True

    def run(self):
        try:
            receiver = MailReceiver(
                imap_host=self.imap_config["host"],
                imap_user=self.imap_config["user"],
                imap_password=self.imap_config["password"],
                imap_port=self.imap_config.get("port", 993),
                use_ssl=self.imap_config.get("use_ssl", True),
            )
            raw_emails = receiver.fetch_recent()
            if self._cancelled:
                return

            results = []
            for raw in raw_emails:
                # 解析发件人姓名和地址（RawEmail.sender 是 "Name <addr>" 格式）
                parsed = email.utils.parseaddr(raw.sender)
                sender_name = parsed[0] or parsed[1]
                sender_addr = parsed[1]

                results.append({
                    "message_id": raw.message_id,
                    "sender": sender_name,
                    "sender_addr": sender_addr,
                    "subject": raw.subject,
                    "body_text": raw.body_text,
                    "received_at": raw.date,
                })

            if not self._cancelled:
                self.emails_fetched.emit(results)

        except Exception as e:
            logger.error("邮箱拉取失败：%s", e)
            if not self._cancelled:
                self.failed.emit(str(e))
```

```python
# shadowtalk/ui/widgets/vertical_tab_bar.py
"""
ShadowTalk 左侧垂直图标 Tab 栏（微信式）。

两个图标按钮上下排列：💬 聊天 / 📧 邮箱。
点击切换时发射 tab_changed(index)，由 MainWindow 同步切换左右面板。
"""
from PySide6.QtWidgets import QWidget, QVBoxLayout, QPushButton, QFrame
from PySide6.QtCore import Signal, Qt, QSize
from PySide6.QtGui import QFont

from shadowtalk.ui.theme import current as theme_current


class VerticalTabBar(QWidget):
    """左侧垂直图标 Tab 栏。"""

    tab_changed = Signal(int)

    def __init__(self, parent=None):
        super().__init__(parent)
        self._buttons: list[QPushButton] = []
        self._current_index = 0
        self._build_ui()

    def _build_ui(self):
        self.setFixedWidth(50)
        self.setObjectName("verticalTabBar")

        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.setSpacing(0)

        # 顶部留白（与品牌栏视觉对齐）
        outer.addSpacing(16)

        # Tab 按钮
        tabs = [
            ("💬", "聊天"),
            ("📧", "邮箱"),
        ]
        for i, (icon, tip) in enumerate(tabs):
            btn = QPushButton(icon)
            btn.setFixedSize(44, 44)
            btn.setCursor(Qt.PointingHandCursor)
            btn.setToolTip(tip)
            btn.setCheckable(True)
            btn.clicked.connect(lambda checked, idx=i: self._on_tab_clicked(idx))
            self._buttons.append(btn)
            outer.addWidget(btn, 0, Qt.AlignHCenter)

        outer.addStretch()

        # 默认选中第一个
        self._buttons[0].setChecked(True)
        self._apply_styles()

    def _on_tab_clicked(self, index: int):
        if index == self._current_index:
            return
        self.set_current(index)
        self.tab_changed.emit(index)

    def set_current(self, index: int):
        """外部程序化切换 Tab（同步左右面板时调用）。"""
        self._current_index = index
        for i, btn in enumerate(self._buttons):
            btn.setChecked(i == index)
        self._apply_styles()

    def _apply_styles(self):
        t = theme_current()
        self.setStyleSheet(f"background-color: {t.SURFACE};")
        for i, btn in enumerate(self._buttons):
            if i == self._current_index:
                # 选中态：accent 底色 + 白色图标
                btn.setStyleSheet(f"""
                    QPushButton {{
                        background-color: {t.ACCENT};
                        color: white;
                        border: none;
                        border-radius: 10px;
                        font-size: 20px;
                    }}
                    QPushButton:hover {{
                        background-color: {t.ACCENT};
                    }}
                """)
            else:
                # 未选中：透明底 + muted 图标
                btn.setStyleSheet(f"""
                    QPushButton {{
                        background-color: transparent;
                        color: {t.MUTED};
                        border: none;
                        border-radius: 10px;
                        font-size: 20px;
                    }}
                    QPushButton:hover {{
                        background-color: {t.BG};
                        color: {t.FG};
                    }}
                """)

    def restyle(self):
        """主题切换后由 MainWindow 调用。"""
        self._apply_styles()
```

```python
# shadowtalk/ui/widgets/mail_folder_list.py
"""
ShadowTalk 邮箱文件夹列表（左栏邮箱模式）。

显示邮箱文件夹（收件箱/已发送/草稿等），点击切换当前文件夹。
文件夹列表由 MailReceiver.list_folders() 获取，首版硬编码常用文件夹。
"""
from PySide6.QtWidgets import (
    QWidget, QVBoxLayout, QListWidget, QListWidgetItem,
    QFrame, QLabel, QPushButton
)
from PySide6.QtCore import Signal, Qt, QSize
from PySide6.QtGui import QFont

from shadowtalk.ui.theme import current as theme_current


# 首版常用文件夹（IMAP 标准名 → 中文显示）
DEFAULT_FOLDERS = [
    ("INBOX", "📥 收件箱"),
    ("Sent", "📤 已发送"),
    ("Drafts", "📝 草稿"),
    ("Trash", "🗑️ 垃圾箱"),
]


class MailFolderListWidget(QWidget):
    """邮箱文件夹列表。"""

    folder_selected = Signal(str)  # 发出 IMAP 文件夹名

    def __init__(self, parent=None):
        super().__init__(parent)
        self._current_folder = "INBOX"
        self._build_ui()

    def _build_ui(self):
        self.setMinimumWidth(200)
        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)

        # ── 头部 ──
        header = QFrame()
        header.setFixedHeight(80)
        header_layout = QHBoxLayout(header)
        header_layout.setContentsMargins(20, 0, 12, 0)
        title = QLabel("📧 邮箱")
        font = QFont()
        font.setPointSize(18)
        font.setBold(True)
        title.setFont(font)
        header_layout.addWidget(title)
        header_layout.addStretch()
        layout.addWidget(header)

        # ── 文件夹列表 ──
        self.list = QListWidget()
        self.list.itemClicked.connect(self._on_item_clicked)
        layout.addWidget(self.list, 1)

        self._load_folders()
        self._apply_styles()

    def _load_folders(self):
        self.list.clear()
        for folder_key, display_name in DEFAULT_FOLDERS:
            item = QListWidgetItem(display_name)
            item.setData(Qt.UserRole, folder_key)
            item.setSizeHint(QSize(0, 44))
            self.list.addItem(item)
            if folder_key == self._current_folder:
                self.list.setCurrentItem(item)

    def _on_item_clicked(self, item):
        folder = item.data(Qt.UserRole)
        self._current_folder = folder
        self.folder_selected.emit(folder)

    def _apply_styles(self):
        t = theme_current()
        self.setStyleSheet(f"background-color: {t.SURFACE};")
        # 头部样式通过 parent 的 restyle 统一处理
        for child in self.findChildren(QFrame):
            child.setStyleSheet(f"background-color: {t.SURFACE};")
        for label in self.findChildren(QLabel):
            label.setStyleSheet(f"color: {t.FG};")
        self.list.setStyleSheet(f"""
            QListWidget {{
                background-color: {t.SURFACE};
                border: none;
                outline: none;
                padding: 4px 8px;
```

```python
        """渲染邮件列表。"""
```

```python
def _on_tab_changed(self, index: int):
    """Tab 栏切换：同步左右面板。"""
    self.left_panel.setCurrentIndex(index)
    self.right_panel.setCurrentIndex(index)
    if index == 1:
        # 切换到邮箱：先读缓存立即渲染，再触发后台刷新
        self.mail_widget.load_folder("INBOX")
        self._on_mail_refresh()

def _on_folder_selected(self, folder: str):
    """文件夹切换：加载该文件夹邮件并触发刷新。"""
    self.mail_widget.load_folder(folder)
    self._on_mail_refresh()

def _on_mail_refresh(self):
    """刷新邮件：后台 IMAP 拉取 → 写缓存 → 重新渲染。"""
    if not Settings.get("mail_imap_host"):
        self.status_bar.showMessage("请先在设置中配置 IMAP 邮箱", 6000)
        return
    # 上一个 worker 在跑则跳过
    if getattr(self, '_mail_fetch_worker', None) and self._mail_fetch_worker.isRunning():
        return
    imap_config = self._get_imap_config()
    if not imap_config:
        return
    from shadowtalk.ui.threads.mail_fetch_worker import MailFetchWorker
    worker = MailFetchWorker(imap_config, "INBOX")
    worker.emails_fetched.connect(self._on_mails_fetched)
    worker.failed.connect(lambda r: self.status_bar.showMessage(f"收信失败：{r}", 6000))
    self._mail_fetch_worker = worker
    self.status_bar.showMessage("正在收信…")
    worker.start()

def _on_mails_fetched(self, emails: list):
    """邮件拉取完成：写缓存 + 重新渲染。"""
    if emails:
        MailCacheRepository.upsert_batch("INBOX", emails)
    self.mail_widget.load_folder("INBOX")
    self.status_bar.showMessage(f"收信完成（{len(emails)} 封）", 4000)

def _on_compose(self):
    """打开写信对话框。"""
    from shadowtalk.ui.widgets.mail_compose_dialog import MailComposeDialog
    dialog = MailComposeDialog(self)
    dialog.send_requested.connect(self._on_send_mail)
    dialog.exec()

def _on_send_mail(self, to: str, subject: str, body: str):
    """发送邮件：SMTP 发送（首版同步，用户主动操作可接受）。"""
    from shadowtalk.core.mail_sender import MailSender, MailSendError
    mail_config = {
        "host": Settings.get("mail_smtp_host"),
        "port": Settings.get_int("mail_smtp_port"),
        "use_tls": Settings.get_bool("mail_smtp_use_tls"),
        "username": Settings.get("mail_smtp_user"),
        "password": Settings.get("mail_smtp_password"),
    }
    if not mail_config["host"]:
        self.status_bar.showMessage("请先在设置中配置 SMTP 服务器", 6000)
        return
    from shadowtalk.core.mail_composer import MailMessage
    mail_msg = MailMessage(
        subject=subject,
        body_text=body,
        message_id=f"user-compose-{to}@shadowtalk.local",
        extra_headers={},
        to=to,
    )
    try:
        sender = MailSender(**mail_config)
        sender.send(mail_msg)
        self.status_bar.showMessage("邮件已发送", 4000)
    except MailSendError as e:
        self.status_bar.showMessage(f"发送失败：{e}", 6000)
```
