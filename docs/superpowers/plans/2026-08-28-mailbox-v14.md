```python
# shadowtalk/core/html_sanitizer.py
"""
HTML 安全过滤器：剥离危险标签和属性，返回安全 HTML。

用于邮箱阅读窗格的 HTML 渲染，防止 XSS 和远程资源加载。
注意：本模块属 core/ 层，纯 Python，禁止 import PySide6。
"""
from bs4 import BeautifulSoup, Comment


# 白名单标签（允许保留）
ALLOWED_TAGS = {
    'p', 'br', 'div', 'span', 'b', 'i', 'u', 's', 'strike',
    'strong', 'em', 'a', 'img', 'table', 'thead', 'tbody', 'tfoot',
    'tr', 'td', 'th', 'caption', 'ul', 'ol', 'li',
    'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
    'blockquote', 'pre', 'code', 'hr',
    'font', 'big', 'small', 'sub', 'sup', 'center',
}

# 白名单属性（按标签）
ALLOWED_ATTRS = {
    '*': {'style', 'align', 'valign', 'width', 'height', 'colspan', 'rowspan', 'bgcolor'},
    'a': {'href', 'title', 'target', 'name'},
    'img': {'src', 'alt', 'title', 'width', 'height'},
    'table': {'border', 'cellpadding', 'cellspacing'},
    'td': {'colspan', 'rowspan', 'width', 'height'},
    'th': {'colspan', 'rowspan', 'width', 'height'},
    'font': {'color', 'size', 'face'},
}

# 危险 CSS 属性值（从 style 中剥离）
DANGEROUS_CSS = {'expression', 'javascript', 'vbscript', 'data:', 'moz-binding'}


def sanitize_html(raw_html: str) -> str:
    """剥离危险标签和属性，返回安全 HTML。"""
    if not raw_html:
        return ''

    soup = BeautifulSoup(raw_html, 'html.parser')

    # 删除注释
    for comment in soup.find_all(string=lambda s: isinstance(s, Comment)):
        comment.extract()

    # 遍历所有标签
    for tag in list(soup.find_all(True)):
        tag_name = tag.name.lower()

        if tag_name not in ALLOWED_TAGS:
            # 危险标签：unwrap（保留子内容）或 extract（删除）
            tag.unwrap()
            continue

        # 过滤属性
        allowed = ALLOWED_ATTRS.get('*', set()) | ALLOWED_ATTRS.get(tag_name, set())
        for attr in list(tag.attrs.keys()):
            attr_lower = attr.lower()
            if attr_lower.startswith('on'):
                del tag[attr]  # 删除所有 on* 事件
                continue
            if attr_lower not in allowed:
                del tag[attr]
                continue
            # 检查 href/src 是否有 javascript: 协议
            if attr_lower in ('href', 'src'):
                value = str(tag[attr]).lower().strip()
                if value.startswith('javascript:') or value.startswith('vbscript:'):
                    del tag[attr]
            # 检查 style 中的危险内容
            if attr_lower == 'style':
                style_val = str(tag[attr]).lower()
                if any(d in style_val for d in DANGEROUS_CSS):
                    del tag[attr]

    return str(soup)
```

```python
def extract_attachments(msg) -> list[dict]:
    """从邮件消息中提取附件列表。

    Args:
        msg: email.message.Message 对象
    Returns:
        附件 dict 列表，每个含 filename/content_type/size/content_id/is_inline/payload
    """
    attachments = []
    for part in msg.walk():
        # 跳过 multipart 容器
        if part.get_content_maintype() == 'multipart':
            continue

        # 跳过正文 part（text/plain 和 text/html）
        content_type = part.get_content_type()
        if content_type in ('text/plain', 'text/html'):
            continue

        filename = part.get_filename() or ''
        content_disposition = str(part.get('Content-Disposition', ''))
        content_id = part.get('Content-ID', '').strip('<>')

        is_inline = ('inline' in content_disposition.lower() and
                     'attachment' not in content_disposition.lower())

        payload = part.get_payload(decode=True) or b''

        attachments.append({
            "filename": filename,
            "content_type": content_type,
            "size": len(payload),
            "content_id": f"<{content_id}>" if content_id else "",
            "is_inline": 1 if is_inline else 0,
            "payload": payload,
        })

    return attachments
```

```python
def _unique_filepath(filepath):
    """如果文件已存在，追加序号。"""
    import os
    if not filepath.exists():
        return filepath
    stem = filepath.stem
    suffix = filepath.suffix
    parent = filepath.parent
    i = 1
    while True:
        candidate = parent / f"{stem}_{i}{suffix}"
        if not candidate.exists():
            return candidate
        i += 1
```

```python
"""
通用邮件发送线程（QThread）。

独立于分身同步的 MailSendWorker，用于邮箱客户端的写信发送。
"""
import logging

from PySide6.QtCore import QThread, Signal

from shadowtalk.core.mail_sender import MailSender

logger = logging.getLogger(__name__)


class MailComposeSendWorker(QThread):
    """通用邮件发送线程。"""

    sent = Signal()              # 发送成功
    failed = Signal(str)         # 失败原因

    def __init__(self, mail_msg, mail_config: dict, parent=None):
        super().__init__(parent)
        self.mail_msg = mail_msg
        self.mail_config = mail_config
        self._cancelled = False

    def cancel(self):
        self._cancelled = True

    def run(self):
        try:
            sender = MailSender(
                smtp_host=self.mail_config["host"],
                smtp_port=int(self.mail_config["port"]),
                use_tls=self.mail_config.get("use_tls", True),
                username=self.mail_config.get("username", ""),
                password=self.mail_config.get("password", ""),
            )
            sender.send(self.mail_msg)
            if not self._cancelled:
                self.sent.emit()
        except Exception as e:
            logger.error("邮件发送失败：%s", e)
            if not self._cancelled:
                self.failed.emit(str(e))
```

```python
def _on_send(self):
    to = self.to_input.text().strip()
    subject = self.subject_input.text().strip()
    html_body = self.body_input.toHtml()
    plain_body = self.body_input.toPlainText().strip()

    if not to or not subject or not plain_body:
        QMessageBox.warning(self, "提示", "请填写收件人、主题和正文")
        return

    # 构建带附件的 MailMessage
    mail_msg = self._build_mail_message(to, subject, html_body, plain_body)
    self.send_requested.emit(mail_msg)  # 信号参数改为 MailMessage 对象

def _build_mail_message(self, to, subject, html_body, plain_body):
    """构建 MIME 邮件（含附件）。"""
    from shadowtalk.core.mail_composer import MailMessage
    from email.mime.multipart import MIMEMultipart
    from email.mime.text import MIMEText
    from email.mime.application import MIMEApplication
    from email.mime.image import MIMEImage
    import os

    if not self._attachments and not self._inline_images:
        # 无附件：简单文本
        return MailMessage(
            subject=subject,
            body_text=plain_body,
            message_id=f"compose-{to}@shadowtalk.local",
            extra_headers={},
            to=to,
        )

    # 有附件：构建 multipart/mixed
    msg = MIMEMultipart('mixed')
    msg['Subject'] = subject
    msg['To'] = to

    # 正文部分（alternative：plain + html）
    body_part = MIMEMultipart('alternative')
    body_part.attach(MIMEText(plain_body, 'plain', 'utf-8'))
    body_part.attach(MIMEText(html_body, 'html', 'utf-8'))
    msg.attach(body_part)

    # 普通附件
    for filepath in self._attachments:
        with open(filepath, 'rb') as f:
            data = f.read()
        att = MIMEApplication(data, Name=os.path.basename(filepath))
        att['Content-Disposition'] = f'attachment; filename="{os.path.basename(filepath)}"'
        msg.attach(att)

    # 返回 MailMessage（body_text 放完整 MIME 字符串）
    return MailMessage(
        subject=subject,
        body_text=plain_body,  # 纯文本 fallback
```

```python
def _on_send_mail(self, mail_msg):
    """异步发送邮件。"""
    from shadowtalk.ui.threads.mail_compose_send_worker import MailComposeSendWorker
    from shadowtalk.core.mail_sender import MailSendError

    mail_config = {
        "host": Settings.get("mail_smtp_host"),
        "port": Settings.get_int("mail_smtp_port"),
        "use_tls": Settings.get_bool("mail_smtp_use_tls"),
        "username": Settings.get("mail_smtp_user"),
        "password": Settings.get("mail_smtp_password"),
    }
    if not mail_config["host"]:
        self.status_bar.showMessage("请先在设置中配置 SMTP 服务器", 6000)
        return

    # TLS 警告
    if not mail_config["use_tls"] and not mail_config["host"].startswith("127.0.0.1"):
        reply = QMessageBox.warning(
            self, "安全警告",
            "当前未启用 TLS 加密，邮件内容将以明文传输。\n仍要发送？",
            QMessageBox.Yes | QMessageBox.No
        )
        if reply == QMessageBox.No:
            return

    worker = MailComposeSendWorker(mail_msg, mail_config)
    worker.sent.connect(lambda: self.status_bar.showMessage("邮件已发送", 4000))
    worker.failed.connect(lambda r: self.status_bar.showMessage(f"发送失败：{r}", 6000))
    self._retired_mail_compose_workers = getattr(self, '_retired_mail_compose_workers', [])
    self._retired_mail_compose_workers.append(worker)
    self.status_bar.showMessage("正在发送…")
    worker.start()
```
